# Dreamscale Labs Favicon Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Generate a complete modern favicon set from the full DSL header mark, place it on white with the approved spacing, and connect it to every public page.

**Architecture:** Use the existing checked-in PNG as the only artwork source, crop its alpha bounds, and composite it deterministically onto square RGB canvases with Pillow. Keep asset validation in the existing Node static-page test, then add one shared metadata contract to the three HTML heads and declare Android icons in a web manifest.

**Tech Stack:** Static HTML, JSON web manifest, Python 3 with Pillow for deterministic image generation, Node.js built-in test runner for asset and markup validation.

## Global Constraints

- Use `assets/logo/dsl-mark.png`, expected at 1302 × 960 pixels with alpha bounds `(129, 103, 1216, 859)`.
- Preserve the complete metallic `DSL` artwork without recoloring, redrawing, sharpening, or adding effects.
- Use an opaque white background in every exported icon.
- Render the standard artwork at exactly 82% of canvas width.
- Render the Android maskable artwork at exactly 72% of canvas width.
- Create ICO frames at 16 × 16, 32 × 32, and 48 × 48.
- Create PNGs at 16 × 16, 32 × 32, 180 × 180, 192 × 192, and 512 × 512, plus a 512 × 512 maskable PNG.
- Add identical favicon metadata to the homepage, blog index, and canonical article.
- Do not change page content, styling, the header logo, or legacy platform metadata.
- Do not add a permanent image-generation dependency or use a generative image model.

## File Structure

- Modify `tests/static-page.test.mjs`: validate PNG headers, RGB opacity mode, ICO frame sizes, manifest contents, and favicon metadata on all public pages.
- Create `favicon.ico`: multi-frame browser fallback.
- Create `favicon-16x16.png`: small browser favicon.
- Create `favicon-32x32.png`: high-density browser favicon.

function readPngMetadata(path) {
  const png = readFileSync(path);
  const signature = Buffer.from([137, 80, 78, 71, 13, 10, 26, 10]);
  assert.deepEqual(png.subarray(0, 8), signature, `${path} must be a PNG`);
  return {
    width: png.readUInt32BE(16),
    height: png.readUInt32BE(20),
    colorType: png[25],
  };
}

function readIcoSizes(path) {
  const ico = readFileSync(path);
  assert.equal(ico.readUInt16LE(0), 0, `${path} has an invalid reserved field`);
  assert.equal(ico.readUInt16LE(2), 1, `${path} must contain icons`);
  const count = ico.readUInt16LE(4);
  return Array.from({ length: count }, (_, index) => {
    const offset = 6 + index * 16;
    return {
      width: ico[offset] || 256,
      height: ico[offset + 1] || 256,
    };
  });
}

Create `.tmp-generate-favicons.py` with:

```python
from pathlib import Path

from PIL import Image


SOURCE = Path("assets/logo/dsl-mark.png")
EXPECTED_SIZE = (1302, 960)
EXPECTED_BOUNDS = (129, 103, 1216, 859)
STANDARD_RATIO = 0.82
MASKABLE_RATIO = 0.72

STANDARD_OUTPUTS = {
    Path("favicon-16x16.png"): 16,
    Path("favicon-32x32.png"): 32,
    Path("apple-touch-icon.png"): 180,
    Path("android-chrome-192x192.png"): 192,
    Path("android-chrome-512x512.png"): 512,
}


def load_mark() -> Image.Image:
    if not SOURCE.is_file():
        raise FileNotFoundError(f"Missing source artwork: {SOURCE}")

    source = Image.open(SOURCE).convert("RGBA")
    if source.size != EXPECTED_SIZE:
        raise ValueError(
            f"Expected source dimensions {EXPECTED_SIZE}, received {source.size}"
        )

    bounds = source.getchannel("A").getbbox()
    if bounds is None:
        raise ValueError("Source artwork contains no non-transparent pixels")
    if bounds != EXPECTED_BOUNDS:
        raise ValueError(
            f"Expected artwork bounds {EXPECTED_BOUNDS}, received {bounds}"
        )

    return source.crop(bounds)


def compose(mark: Image.Image, size: int, width_ratio: float) -> Image.Image:
    target_width = round(size * width_ratio)
    target_height = round(mark.height * target_width / mark.width)
    resized = mark.resize(
        (target_width, target_height),
        Image.Resampling.LANCZOS,
    )
    canvas = Image.new("RGB", (size, size), "white")
    position = (
        (size - target_width) // 2,
        (size - target_height) // 2,
    )
    canvas.paste(resized, position, resized)
    return canvas


def main() -> None:
    mark = load_mark()

    for output, size in STANDARD_OUTPUTS.items():
        compose(mark, size, STANDARD_RATIO).save(
            output,
            format="PNG",
            optimize=True,
        )

    compose(mark, 512, MASKABLE_RATIO).save(
        "android-chrome-maskable-512x512.png",
        format="PNG",
        optimize=True,
    )

    compose(mark, 256, STANDARD_RATIO).save(
        "favicon.ico",
        format="ICO",
        sizes=[(16, 16), (32, 32), (48, 48)],
    )


if __name__ == "__main__":
    main()
```
